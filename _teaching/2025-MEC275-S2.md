### Notas — Tabla filtrable
Ingrese su rol/ID cuando se le pida y la tabla mostrará solo las filas asociadas.

<div id="grades-app">
  <p>Cargando notas...</p>
</div>

<script src="https://cdn.jsdelivr.net/npm/papaparse@5.3.2/papaparse.min.js"></script>
<script>
  (function(){
    // URL pública en formato CSV (reemplazada a partir de tu pubhtml)
    const CSV_URL = 'https://docs.google.com/spreadsheets/d/e/2PACX-1vQsgbMMB8KrN9rhMVDtS68HxOeGlne7BD25ig75ExZHrPT5V6NxQlS6Q36cBkKu0w/pub?gid=2096219777&single=true&output=csv';
    const ROLE_COL = 'ROL_USM'; // columna que contiene el rol/ID
    const NUM_COLS_AFTER_ROLE = 11; // cuántas columnas a la derecha mostrar

    const container = document.getElementById('grades-app');

    function renderTable(headers, rows) {
      const table = document.createElement('table');
      table.style.borderCollapse = 'collapse';
      table.style.width = '100%';
      const thead = document.createElement('tr');
      headers.forEach(h => {
        const th = document.createElement('th');
        th.textContent = h;
        th.style.border = '1px solid #ccc';
        th.style.padding = '6px';
        th.style.textAlign = 'left';
        thead.appendChild(th);
      });
      table.appendChild(thead);
      rows.forEach(row => {
        const tr = document.createElement('tr');
        headers.forEach(h => {
          const td = document.createElement('td');
          td.textContent = row[h] || '';
          td.style.border = '1px solid #eee';
          td.style.padding = '6px';
          tr.appendChild(td);
        });
        table.appendChild(tr);
      });
      container.innerHTML = '';
      container.appendChild(table);
    }

    async function loadAndFilter() {
      const role = prompt('Ingrese su rol o identificador (ej: ROL_USM):');
      if (!role) { container.innerHTML = '<p>No se ingresó rol.</p>'; return; }
      container.innerHTML = '<p>Consultando datos...</p>';
      try {
        const res = await fetch(CSV_URL);
        if (!res.ok) throw new Error('No se pudo obtener la hoja. Verifique la URL y que esté publicada como CSV.');
        const csv = await res.text();
        const parsed = Papa.parse(csv, { header: true, skipEmptyLines: true });
        const data = parsed.data;
        const headers = parsed.meta.fields || (data[0] ? Object.keys(data[0]) : []);
        if (!headers || headers.length === 0) throw new Error('No se pudieron leer los encabezados del CSV.');

        const roleIndex = headers.indexOf(ROLE_COL);
        if (roleIndex === -1) throw new Error('No se encontró la columna "' + ROLE_COL + '" en la hoja.');

        const selectedCols = headers.slice(roleIndex + 1, roleIndex + 1 + NUM_COLS_AFTER_ROLE);
        const displayHeaders = [ROLE_COL].concat(selectedCols);

        const filtered = data.filter(r => (r[ROLE_COL] || '').toString().trim().toLowerCase() === role.trim().toLowerCase());
        if (filtered.length === 0) {
          container.innerHTML = `<p>No hay datos para "${role}".</p>`;
          return;
        }

        // Map rows a solo las columnas seleccionadas (mantener encabezados)
        const rowsToShow = filtered.map(r => {
          const obj = {};
          displayHeaders.forEach(h => obj[h] = r[h] || '');
          return obj;
        });

        renderTable(displayHeaders, rowsToShow);
      } catch (err) {
        container.innerHTML = '<p>Error: ' + err.message + '</p>';
        console.error(err);
      }
    }

    document.addEventListener('DOMContentLoaded', loadAndFilter);
  })();
</script>
