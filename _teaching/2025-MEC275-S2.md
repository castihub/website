  <div style="margin-bottom:8px;">
    <input id="role-input" type="text" inputmode="numeric" maxlength="9" placeholder="Ingrese su ROL USM (9 dígitos)" style="padding:6px; width:220px; color:#000; background:#fff; border:1px solid #ccc;" />
    <button id="role-btn" style="padding:6px 10px;">Buscar</button>

  <div id="grades-result" style="color:#000;"><p>Esperando acción...</p></div>

<script>
  (function(){
    const CSV_URL = 'https://docs.google.com/spreadsheets/d/e/2PACX-1vQsgbMMB8KrN9rhMVDtS68HxOeGlne7BD25ig75ExZHrPT5V6NxQlS6Q36cBkKu0w/pub?gid=2096219777&single=true&output=csv';
    const ROLE_COL = 'ROL_USM';
    const NUM_COLS_AFTER_ROLE = 11;

    const resultDiv = document.getElementById('grades-result');
    const input = document.getElementById('role-input');
    const btn = document.getElementById('role-btn');

    function show(msg){ resultDiv.innerHTML = '<pre style="color:#000; white-space:pre-wrap;">' + msg + '</pre>'; console.log(msg); }

    input.addEventListener('input', function(){
      this.value = this.value.replace(/\D/g,'').slice(0,9);
    });

    function parseCSV(text){
      const rows = [];
      let i = 0, cur = '', row = [], inQuotes = false;
      while (i < text.length) {
        const ch = text[i];
        if (ch === '"') {
          if (inQuotes && text[i+1] === '"') { cur += '"'; i += 2; continue; }
          inQuotes = !inQuotes; i++; continue;
        }
        if (ch === ',' && !inQuotes) { row.push(cur); cur = ''; i++; continue; }
        if ((ch === '\n' || ch === '\r') && !inQuotes) {
          if (ch === '\r' && text[i+1] === '\n') i++;
          row.push(cur); rows.push(row); row = []; cur = ''; i++; continue;
        }
        cur += ch; i++;
      }
      if (cur !== '' || row.length > 0) { row.push(cur); rows.push(row); }
      return rows;
    }

    function buildObjectsFromRows(rows){
      const headerRowIndex = rows.findIndex(r => r.some(c => (c||'').toString().trim().toLowerCase() === ROLE_COL.toLowerCase()));
      return { headerRowIndex, rows };
    }

    function renderTable(headers, rows){
      const table = document.createElement('table');
      table.style.borderCollapse = 'collapse';
      table.style.width = '100%';
      table.style.marginTop = '8px';
      table.style.color = '#000';
      const thead = document.createElement('tr');
      headers.forEach(h => {
        const th = document.createElement('th');
        th.textContent = h;
        th.style.border = '1px solid #ccc';
        th.style.padding = '6px';
        th.style.textAlign = 'left';
        thead.appendChild(th);
      });
      table.appendChild(thead);
      rows.forEach(row => {
        const tr = document.createElement('tr');
        headers.forEach(h => {
          const td = document.createElement('td');
          td.textContent = row[h] || '';
          td.style.border = '1px solid #eee';
          td.style.padding = '6px';
          tr.appendChild(td);
        });
        table.appendChild(tr);
      });
      resultDiv.innerHTML = '';
      resultDiv.appendChild(table);
    }

    async function fetchAndShow(role){
      try {
        if (!/^\d{9}$/.test(role)) { show('El ROL debe tener exactamente 9 dígitos.'); return; }
        show('Iniciando consulta...\nRole: ' + role);
        const res = await fetch(CSV_URL);
        show('Fetch status: ' + res.status + ' ' + res.statusText);
        if (!res.ok) throw new Error('HTTP ' + res.status);
        const csv = await res.text();
        show('CSV length: ' + csv.length + '\nCSV preview:\n' + csv.slice(0,300));
        if (csv.trim().startsWith('<')) throw new Error('El recurso parece HTML, no CSV. Verifica output=csv.');

        const rows = parseCSV(csv);
        show('Filas parsed: ' + rows.length + '\nPrimera 3 filas:\n' + rows.slice(0,3).map(r=>r.join('|')).join('\n'));
        const built = buildObjectsFromRows(rows);
        console.log('buildObjectsFromRows result:', built);

        const headerRowIndex = built.headerRowIndex;
        show('headerRowIndex: ' + headerRowIndex);
        if (headerRowIndex === -1) throw new Error('No se encontró fila con el encabezado "' + ROLE_COL + '".');

        const headersRaw = rows[headerRowIndex].map(h => (h||'').toString().replace(/^\uFEFF/,'').trim());
        show('Headers detected: ' + headersRaw.join(', '));
        const idx = headersRaw.findIndex(h => (h||'').toLowerCase() === ROLE_COL.toLowerCase());
        show('ROL_USM index in headers: ' + idx);

        if (idx === -1) throw new Error('No se encontró la columna exacta "' + ROLE_COL + '". Encabezados: ' + headersRaw.join(', '));

        const dataRows = rows.slice(headerRowIndex + 1);
        const objs = dataRows.map(row => {
          const obj = {};
          for (let i=0;i<headersRaw.length;i++) obj[headersRaw[i]] = (row[i] !== undefined) ? row[i].toString().trim() : '';
          return obj;
        });
        show('Filas de datos: ' + objs.length);

        const filtered = objs.filter(r => ((r[headersRaw[idx]]||'').toString().trim() === role));
        show('Filas que coinciden con role: ' + filtered.length + (filtered.length ? '\nPrimera coincidente: ' + JSON.stringify(filtered[0]) : ''));
        if (filtered.length === 0) { return; }

        const selectedCols = headersRaw.slice(idx + 1, idx + 1 + NUM_COLS_AFTER_ROLE);
        const displayHeaders = [headersRaw[idx]].concat(selectedCols);
        const rowsToShow = filtered.map(r => {
          const obj = {};
          displayHeaders.forEach(h => obj[h] = r[h] || '');
          return obj;
        });
        renderTable(displayHeaders, rowsToShow);
      } catch (err) {
        console.error(err);
        show('Error: ' + err.message);
      }
    }

    btn.addEventListener('click', function(){
      const role = (input.value || '').trim();
      fetchAndShow(role);
    });

    document.addEventListener('DOMContentLoaded', function(){
      const params = new URLSearchParams(location.search);
      const q = params.get('role') || params.get('rol');
      if (q) { input.value = q.replace(/\D/g,'').slice(0,9); fetchAndShow(input.value); }
    });
  })();
</script>
