### Notas — Tabla filtrable
Ingrese su rol/ID (9 dígitos) y presione "Buscar" para ver sus calificaciones (se muestran 11 columnas a la derecha de ROL_USM).

  <div style="margin-bottom:8px;">
    <input id="role-input" type="text" inputmode="numeric" maxlength="9" placeholder="Ingrese su ROL_USM (9 dígitos)" style="padding:6px; width:220px; color:#000; background:#fff; border:1px solid #ccc;" />
    <button id="role-btn" style="padding:6px 10px;">Buscar</button>
  </div>
  <div id="grades-result" style="color:#000;"><p>Esperando acción...</p></div>

<script>
  (function(){
    const CSV_URL = 'https://docs.google.com/spreadsheets/d/e/2PACX-1vQsgbMMB8KrN9rhMVDtS68HxOeGlne7BD25ig75ExZHrPT5V6NxQlS6Q36cBkKu0w/pub?gid=2096219777&single=true&output=csv';
    const ROLE_COL = 'ROL_USM';
    const NUM_COLS_AFTER_ROLE = 11;

    const resultDiv = document.getElementById('grades-result');
    const input = document.getElementById('role-input');
    const btn = document.getElementById('role-btn');

    function show(msg){ resultDiv.innerHTML = '<p style="color:#000;">' + msg + '</p>'; console.log(msg); }

    // Limitar input a dígitos y max 9
    input.addEventListener('input', function(){
      this.value = this.value.replace(/\D/g,'').slice(0,9);
    });

    // Simple CSV parser (no eval). Maneja campos entrecomillados y CR/LF.
    function parseCSV(text){
      const rows = [];
      let i = 0, cur = '', row = [], inQuotes = false;
      while (i < text.length) {
        const ch = text[i];
        if (ch === '"') {
          if (inQuotes && text[i+1] === '"') { cur += '"'; i += 2; continue; }
          inQuotes = !inQuotes; i++; continue;
        }
        if (ch === ',' && !inQuotes) { row.push(cur); cur = ''; i++; continue; }
        if ((ch === '\n' || ch === '\r') && !inQuotes) {
          if (ch === '\r' && text[i+1] === '\n') i++;
          row.push(cur); rows.push(row); row = []; cur = ''; i++; continue;
        }
        cur += ch; i++;
      }
      if (cur !== '' || row.length > 0) { row.push(cur); rows.push(row); }
      return rows;
    }

    function buildObjectsFromRows(rows){
      // buscar índice de fila que contiene el encabezado con ROL_USM
      const headerRowIndex = rows.findIndex(r => r.some(c => (c||'').toString().trim().toLowerCase() === ROLE_COL.toLowerCase()));
      if (headerRowIndex === -1) return { error: 'No se encontró fila de encabezados con "' + ROLE_COL + '". Encabezados detectados (primeras filas): ' + rows.slice(0,3).map(r=>r.join('|')).join(' --- ') };
      const headersRaw = rows[headerRowIndex].map(h => (h||'').toString().replace(/^\uFEFF/,'').trim());
      const dataRows = rows.slice(headerRowIndex + 1);
      const objs = dataRows.reduce((acc, row) => {
        // saltar filas vacías
        if (row.length === 1 && (row[0]||'').trim() === '') return acc;
        const obj = {};
        for (let c = 0; c < headersRaw.length; c++) {
          obj[headersRaw[c]] = (row[c] !== undefined) ? row[c].toString().trim() : '';
        }
        acc.push(obj);
        return acc;
      }, []);
      return { headers: headersRaw, data: objs };
    }

    function renderTable(headers, rows){
      const table = document.createElement('table');
      table.style.borderCollapse = 'collapse';
      table.style.width = '100%';
      table.style.marginTop = '8px';
      table.style.color = '#000';
      const thead = document.createElement('tr');
      headers.forEach(h => {
        const th = document.createElement('th');
        th.textContent = h;
        th.style.border = '1px solid #ccc';
        th.style.padding = '6px';
        th.style.textAlign = 'left';
        thead.appendChild(th);
      });
      table.appendChild(thead);
      rows.forEach(row => {
        const tr = document.createElement('tr');
        headers.forEach(h => {
          const td = document.createElement('td');
          td.textContent = row[h] || '';
          td.style.border = '1px solid #eee';
          td.style.padding = '6px';
          tr.appendChild(td);
        });
        table.appendChild(tr);
      });
      resultDiv.innerHTML = '';
      resultDiv.appendChild(table);
    }

    async function fetchAndShow(role){
      try {
        if (!/^\d{9}$/.test(role)) { show('El ROL debe tener exactamente 9 dígitos.'); return; }
        show('Consultando datos...');
        const res = await fetch(CSV_URL);
        if (!res.ok) throw new Error('Respuesta HTTP ' + res.status);
        const csv = await res.text();
        if (csv.trim().startsWith('<')) throw new Error('El recurso no parece ser CSV. Verifica que la hoja esté "Publicar en la web" en formato CSV (output=csv).');
        const rows = parseCSV(csv);
        const parsed = buildObjectsFromRows(rows);
        if (parsed.error) throw new Error(parsed.error);
        const headers = parsed.headers;
        // buscar índice de ROL_USM
        const idx = headers.findIndex(h => (h||'').toLowerCase() === ROLE_COL.toLowerCase());
        if (idx === -1) throw new Error('No se encontró la columna "' + ROLE_COL + '". Encabezados detectados: ' + headers.join(', '));
        const selectedCols = headers.slice(idx + 1, idx + 1 + NUM_COLS_AFTER_ROLE);
        const displayHeaders = [headers[idx]].concat(selectedCols);
        const filtered = parsed.data.filter(r => ((r[headers[idx]]||'').toString().trim().toLowerCase() === role.trim().toLowerCase()));
        if (filtered.length === 0) { show(`No hay datos para "${role}".`); return; }
        const rowsToShow = filtered.map(r => {
          const obj = {};
          displayHeaders.forEach(h => obj[h] = r[h] || '');
          return obj;
        });
        renderTable(displayHeaders, rowsToShow);
      } catch (err) {
        console.error(err);
        show('Error: ' + err.message);
      }
    }

    btn.addEventListener('click', function(){
      const role = (input.value || '').trim();
      fetchAndShow(role);
    });

    // Soporte query param ?role=XXXXXXXXX
    document.addEventListener('DOMContentLoaded', function(){
      const params = new URLSearchParams(location.search);
      const q = params.get('role') || params.get('rol');
      if (q) { input.value = q.replace(/\D/g,'').slice(0,9); fetchAndShow(input.value); }
    });
  })();
</script>
