Se abrirá un prompt para pedir tu rol y se mostrarán solo las filas asociadas.

  <p>Cargando aplicación de notas... si no aparece el prompt, recarga la página.</p>

<!-- Dependencia para parsear CSV robustamente -->
<script src="https://cdn.jsdelivr.net/npm/papaparse@5.3.2/papaparse.min.js"></script>
<script>
  (function(){
    // 1) Pegar aquí la URL pública CSV de Google Sheets:
    const CSV_URL = 'https://docs.google.com/spreadsheets/d/e/2PACX-1vQsgbMMB8KrN9rhMVDtS68HxOeGlne7BD25ig75ExZHrPT5V6NxQlS6Q36cBkKu0w/pub?gid=69786769&single=true&output=csv';

    const container = document.getElementById('grades-app');

    function renderTable(headers, rows) {
      const table = document.createElement('table');
      table.style.borderCollapse = 'collapse';
      table.style.width = '100%';
      const thRow = document.createElement('tr');
      headers.forEach(h => {
        const th = document.createElement('th');
        th.textContent = h;
        th.style.border = '1px solid #ccc';
        th.style.padding = '6px';
        th.style.textAlign = 'left';
        thRow.appendChild(th);
      });
      table.appendChild(thRow);
      rows.forEach(row => {
        const tr = document.createElement('tr');
        headers.forEach(h => {
          const td = document.createElement('td');
          td.textContent = row[h] || '';
          td.style.border = '1px solid #eee';
          td.style.padding = '6px';
          tr.appendChild(td);
        });
        table.appendChild(tr);
      });
      container.innerHTML = '';
      container.appendChild(table);
    }

    async function loadAndFilter() {
      try {
        const role = prompt('Ingrese su rol (por ejemplo: alumno, ayudante, docente) o su identificador:');
        if (!role) {
          container.innerHTML = '<p>No se ingresó rol.</p>';
          return;
        }
        container.innerHTML = '<p>Consultando datos...</p>';
        const res = await fetch(CSV_URL);
        if (!res.ok) throw new Error('No se pudo obtener la hoja. Verifique la URL y que esté publicada.');
        const csv = await res.text();
        const parsed = Papa.parse(csv, { header: true, skipEmptyLines: true });
        const data = parsed.data;
        // Ajusta aquí el nombre de la columna que contiene el "rol" en tu sheet (por ejemplo: "rol" o "tipo")
        const ROLE_COL = 'ROL_USM'; // cambiar si tu columna se llama distinto
        const filtered = data.filter(r => (r[ROLE_COL] || '').toString().trim().toLowerCase() === role.trim().toLowerCase());
        if (filtered.length === 0) {
          container.innerHTML = `<p>No hay datos para el rol "${ROL_USM}".</p>`;
          return;
        }
        renderTable(Object.keys(filtered[0]), filtered);
      } catch (err) {
        container.innerHTML = '<p>Error: ' + err.message + '</p>';
        console.error(err);
      }
    }

    // Ejecutar al cargar
    document.addEventListener('DOMContentLoaded', loadAndFilter, false);
  })();
</script>
